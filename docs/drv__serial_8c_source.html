66int drv_serial_open(drv_serial_t *ctx,
67 const char *port,
68 uint32_t baudrate)
69{
70 ctx->handle = CreateFileA(port,
71 GENERIC_READ | GENERIC_WRITE,
72 0,
73 NULL,
74 OPEN_EXISTING,
75 0,
76 NULL);
77
78 if (ctx->handle == INVALID_HANDLE_VALUE)
79 return -1;
80
81 DCB dcb = {0};
82 dcb.DCBlength = sizeof(dcb);
83
84 /* Retrieve current configuration */
85 if (!GetCommState(ctx->handle, &dcb))
86 return -1;
87
88 /* Configure serial parameters */
89 dcb.BaudRate = baudrate;
90 dcb.ByteSize = 8;
91 dcb.StopBits = ONESTOPBIT;
92 dcb.Parity = NOPARITY;
93
94 if (!SetCommState(ctx->handle, &dcb))
95 return -1;
96
97 return 0;
98}

201int drv_serial_open(drv_serial_t *ctx,
202 const char *port,
203 uint32_t baudrate)
204{
205 if(!ctx || !port)
206 return -1;
207
208 /* Open serial port */
209 ctx->fd = open(port, O_RDWR | O_NOCTTY | O_SYNC);
210 if(ctx->fd < 0)
211 return -1;
212
213 struct termios tty;
214 memset(&tty, 0, sizeof(tty));
215
216 if(tcgetattr(ctx->fd, &tty) != 0)
217 return -1;
218
219 /* Set baud rate (currently fixed to 115200) */
220 cfsetospeed(&tty, B115200);
221 cfsetispeed(&tty, B115200);
222
223 /* 8-bit chars */
224 tty.c_cflag = (tty.c_cflag & ~CSIZE) | CS8;
225
226 /* Enable receiver and set local mode */
227 tty.c_cflag |= (CLOCAL | CREAD);
228
229 /* Disable parity */
230 tty.c_cflag &= ~PARENB;
231
232 /* 1 stop bit */
233 tty.c_cflag &= ~CSTOPB;
234
235 /* No hardware flow control */
236 tty.c_cflag &= ~CRTSCTS;
237
238 /* Raw input */
239 tty.c_iflag &= ~(IXON | IXOFF | IXANY);
240 tty.c_iflag &= ~(IGNBRK | BRKINT | PARMRK | ISTRIP | INLCR | IGNCR | ICRNL);
241
242 /* Raw output */
243 tty.c_oflag = 0;
244
245 /* Disable canonical mode, echo, signals */
246 tty.c_lflag = 0;
247
248 /*
249 Read behavior
250 VMIN = 0 → read returns immediately
251 VTIME = 1 → 100ms timeout
252 This is important so your RX thread can exit.
253 */
254 tty.c_cc[VMIN] = 0;
255 tty.c_cc[VTIME] = 1;
256
257 if(tcsetattr(ctx->fd, TCSANOW, &tty) != 0)
258 return -1;
259
260 return 0;
261}
