189{
190 /* ------------------------------------------------------------------
191 ST_INIT → ST_SEND_RESET
192 First event in every session. Send CMD_RESET_REQ to the target.
193 ------------------------------------------------------------------ */
194 {
195 &ST_INIT_STATE,
196 EVT_START,
197 &ST_SEND_RESET_STATE,
198 act_send_reset
199 },
200
201 /* ------------------------------------------------------------------
202 ST_SEND_RESET → ST_BUILD_PIPELINE
203 Target replied with hardware parameters. Parse and store them,
204 then build the pipeline.
205 ------------------------------------------------------------------ */
206 {
207 &ST_SEND_RESET_STATE,
208 EVT_TARGET_INFO,
209 &ST_BUILD_PIPE_STATE,
210 act_target_info
211 },
212
213 /* ------------------------------------------------------------------
214 ST_BUILD_PIPELINE → ST_SEND_WINDOW
215 Pipeline built. Move to the first segment send immediately.
216 ------------------------------------------------------------------ */
217 {
218 &ST_BUILD_PIPE_STATE,
219 EVT_START,
220 &ST_SEND_WINDOW_STATE,
221 act_build_pipeline
222 },
223
224 /* ------------------------------------------------------------------
225 ST_SEND_WINDOW — three outgoing transitions.
226
227 Row 1 — EVT_START:
228 Used for the very first segment of each sector (dispatched by
229 act_build_pipeline and act_next_sector). Calls act_send_window
230 to transmit one segment, then the FSM waits for EVT_SEG_ACK.
231
232 Row 2 — EVT_SEG_ACK (KEY FIX):
233 Wired DIRECTLY to act_send_window — no intermediate act_seg_ack
234 hop. act_send_window either sends the next segment (loop) or
235 detects sector exhaustion and dispatches EVT_SECTOR_END (exit
236 to ST_VERIFY). This eliminates the two-event queue burst of
237 the previous design and makes every ACK→send transition atomic.
238
239 Row 3 — EVT_SECTOR_END:
240 All segments of the sector were sent and acknowledged.
241 Move to verification.
242 ------------------------------------------------------------------ */
243 {
244 &ST_SEND_WINDOW_STATE,
245 EVT_START,
246 &ST_SEND_WINDOW_STATE,
247 act_send_window
248 },
249 {
250 &ST_SEND_WINDOW_STATE,
251 EVT_SEG_ACK,
252 &ST_SEND_WINDOW_STATE,
253 act_send_window
254 },
255 {
256 &ST_SEND_WINDOW_STATE,
257 EVT_SECTOR_END,
258 &ST_VERIFY_STATE,
259 act_crc_verify
260 },
261
262 /* ------------------------------------------------------------------
263 ST_VERIFY — two outgoing transitions.
264
265 Row 1 — EVT_CRC_OK:
266 Target wrote and verified the sector. Advance the sector cursor
267 or signal that all sectors are done.
268
269 Row 2 — EVT_CRC_NACK:
270 Write failed. Reset offset back to 0 for the current sector,
271 increment the retry counter, and loop back to ST_SEND_WINDOW
272 to retransmit every segment of the sector from scratch.
273 ------------------------------------------------------------------ */
274 {
275 &ST_VERIFY_STATE,
276 EVT_CRC_OK,
277 &ST_NEXT_SECTOR_STATE,
278 act_next_sector
279 },
280 {
281 &ST_VERIFY_STATE,
282 EVT_CRC_NACK,
283 &ST_SEND_WINDOW_STATE,
284 act_crc_nack
285 },
286
287 /* ------------------------------------------------------------------
288 ST_NEXT_SECTOR — two outgoing transitions.
289
290 Row 1 — EVT_START:
291 More sectors remain. Loop back and begin sending the next one.
292
293 Row 2 — EVT_ALL_SECTORS_DONE:
294 Last sector just finished. Send the application start command.
295 ------------------------------------------------------------------ */
296 {
297 &ST_NEXT_SECTOR_STATE,
298 EVT_START,
299 &ST_SEND_WINDOW_STATE,
300 act_send_window
301 },
302 {
303 &ST_NEXT_SECTOR_STATE,
304 EVT_ALL_SECTORS_DONE,
305 &ST_APP_JUMP_STATE,
306 act_app_jump
307 },
308
309 /* ------------------------------------------------------------------
310 ST_APP_JUMP → ST_DONE
311 Target confirmed the application jump. Log completion and stop.
312 ------------------------------------------------------------------ */
313 {
314 &ST_APP_JUMP_STATE,
315 EVT_APP_ACK,
316 &ST_DONE_STATE,
317 act_done
318 },
319};
