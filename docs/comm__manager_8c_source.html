114void* comm_rx_thread(void *arg)
115{
116 int32_t *thread_running_flag = (int32_t *)arg;
117
118 while (*thread_running_flag)
119 {
120 /* ---- Block until a frame arrives or the thread is stopped --- */
121 int ret = transport_receive(&receive_info_packet, thread_running_flag);
122
123 if (ret > 0)
124 {
125 /* ---- Valid packet received — queue it for the FSM ------- */
126
127 comm_packet_t *packet_ptr =
128 (comm_packet_t *)malloc(sizeof(comm_packet_t));
129
130 if (packet_ptr == NULL)
131 {
132 /* Out of memory — skip this packet and keep running.
133 The FSM will time out waiting for the ACK and the
134 target will retransmit. */
135 printf("[ RX ] malloc failed — dropping cmd=0x%02X\n",
136 receive_info_packet.command);
137 continue;
138 }
139
140 /* Copy the stack packet into the heap allocation */
141 memcpy(packet_ptr, &receive_info_packet, sizeof(comm_packet_t));
142
143 /* Transfer ownership to the queue */
144 queue_push(&handle_queue_receive_packets, packet_ptr);
145
146 // printf("[ RX ] Queued cmd=0x%02X len=%u\n",
147 // packet_ptr->command, packet_ptr->length);
148 }
149 else if (ret == -5)
150 {
151 /* CRC mismatch — transport_receive already printed details.
152 Parser has been reset; just loop back and wait for the
153 next frame. The host will time out and resend its last
154 command if needed. */
155 printf("[ RX ] CRC error — waiting for next frame\n");
156 }
157 else if (ret == -3)
158 {
159 /* Framing error (payload too large) — same recovery. */
160 printf("[ RX ] Framing error — waiting for next frame\n");
161 }
162 /* ret == -1 means thread was stopped — the while condition
163 will catch that on the next iteration and exit cleanly. */
164 }
165 return NULL;
166}
