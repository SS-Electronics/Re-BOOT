116void queue_push(queue_t *q, void *msg)
117{
118 queue_node_t *node = malloc(sizeof(queue_node_t));
119
120 node->data = msg;
121 node->next = NULL;
122
123 mutex_lock(&q->mutex);
124
125 if (q->tail)
126 q->tail->next = node;
127 else
128 q->head = node;
129
130 q->tail = node;
131 q->size++;
132
133 /* wake one waiting consumer */
134 cond_signal(&q->cond);
135
136 mutex_unlock(&q->mutex);
137}

152void* queue_pop(queue_t *q)
153{
154 mutex_lock(&q->mutex);
155
156 /* wait until queue contains data */
157 while (q->size == 0)
158 cond_wait(&q->cond, &q->mutex);
159
160 queue_node_t *node = q->head;
161
162 q->head = node->next;
163
164 if (!q->head)
165 q->tail = NULL;
166
167 q->size--;
168
169 mutex_unlock(&q->mutex);
170
171 void *data = node->data;
172
173 free(node);
174
175 return data;
176}
