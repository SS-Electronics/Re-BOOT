99static sector_pipeline_t *find_sector(pipeline_builder_t *pb, uint32_t addr)
100{
101 uint32_t base = sector_base(addr, pb->sector_size);
102
103 /* Search existing sectors for a matching base address */
104 for (uint32_t i = 0; i < pb->sector_count; i++)
105 {
106 if (pb->sectors[i].base_addr == base)
107 return &pb->sectors[i];
108 }
109
110 /* No match — allocate a new sector slot at the end of the array */
111 pb->sectors = realloc(
112 pb->sectors,
113 sizeof(sector_pipeline_t) * (pb->sector_count + 1)
114 );
115
116 sector_pipeline_t *s = &pb->sectors[pb->sector_count++];
117
118 s->base_addr = base;
119 s->size = pb->sector_size;
120
121 /* Allocate and initialise data and validity buffers */
122 s->data = malloc(pb->sector_size);
123 s->valid = malloc(pb->sector_size);
124
125 /* Pre-fill data with 0xFF (erased flash state) */
126 memset(s->data, 0xFF, pb->sector_size);
127
128 /* Mark every byte as invalid until a HEX record writes it */
129 memset(s->valid, 0, pb->sector_size);
130
131 return s;
132}

144static void insert_record(pipeline_builder_t *pb, hex_record_t *rec)
145{
146 /* Skip non-data records (extended address, EOF, etc.) */
147 if (rec->type != 0x00)
148 return;
149
150 for (uint32_t i = 0; i < rec->length; i++)
151 {
152 uint32_t addr = rec->address + i;
153
154 /* Locate (or create) the sector that owns this address */
155 sector_pipeline_t *sec = find_sector(pb, addr);
156
157 /* Offset of this byte within the sector */
158 uint32_t off = addr - sec->base_addr;
159
160 sec->data[off] = rec->data[i];
161 sec->valid[off] = 1;
162 }
163}

183void pipeline_build(
184 pipeline_builder_t *pb,
185 hex_record_t *records,
186 uint32_t count,
187 uint32_t sector_size)
188{
189 pb->sector_size = sector_size;
190 pb->sector_count = 0;
191 pb->sectors = NULL;
192
193 for (uint32_t i = 0; i < count; i++)
194 insert_record(pb, &records[i]);
195}

220int pipeline_next_segment(
221 pipeline_builder_t *pb,
222 uint32_t sector_index,
223 uint32_t *offset,
224 uint32_t seg,
225 uint32_t *addr,
226 uint8_t *data)
227{
228 sector_pipeline_t *sec = &pb->sectors[sector_index];
229
230 /* Skip leading invalid (unprogrammed) bytes */
231 while (*offset < sec->size)
232 {
233 if (sec->valid[*offset])
234 break;
235
236 (*offset)++;
237 }
238
239 /* Sector exhausted — signal the caller to move to verify */
240 if (*offset >= sec->size)
241 return -1;
242
243 /* Clamp segment length to what remains in the sector */
244 uint32_t len = seg;
245
246 if (*offset + len > sec->size)
247 len = sec->size - *offset;
248
249 /* Copy segment data and record the starting flash address */
250 memcpy(data, &sec->data[*offset], len);
251 *addr = sec->base_addr + *offset;
252 *offset += len;
253
254 return (int)len;
255}

271uint32_t pipeline_sector_crc(pipeline_builder_t *pb, uint32_t sector_index)
272{
273 sector_pipeline_t *sec = &pb->sectors[sector_index];
274
275 uint32_t crc = 0xFFFFFFFFu;
276
277 for (uint32_t i = 0; i < sec->size; i++)
278 {
279 crc ^= (uint32_t)sec->data[i];
280
281 /* Process each bit with the reflected CRC32 polynomial */
282 for (uint8_t bit = 0; bit < 8; bit++)
283 {
284 if (crc & 1u)
285 crc = (crc >> 1) ^ 0xEDB88320u; /* standard CRC32 poly */
286 else
287 crc >>= 1;
288 }
289 }
290
291 /* Apply final XOR to produce the standard CRC32 result */
292 return crc ^ 0xFFFFFFFFu;
293}
