57int fileio_open(fileio_t *f, const char *path, fileio_mode_t mode)

136int fileio_printf(fileio_t *f, const char *fmt, ...)
137{
138 if(!f || !f->handle)
139 return -1;
140
141 char buffer[1024];
142
143 va_list args;
144 va_start(args, fmt);
145
146 int len = vsnprintf(buffer, sizeof(buffer), fmt, args);
147
148 va_end(args);
149
150 if(len <= 0)
151 return -1;
152
153 return fwrite(buffer, 1, len, f->handle);
154}
