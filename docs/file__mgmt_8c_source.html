53int parse_arguments(int argc, char *argv[], cmd_args_t *args)
54{
55 int i;
56
57 /* Initialize defaults */
58 args->file_path = NULL;
59 args->node_id = -1;
60 args->n_retry = 0;
61 args->reset = 0;
62
63 for (i = 1; i < argc; i++)
64 {
65 if (strcmp(argv[i], "-f") == 0)
66 {
67 if (i + 1 < argc)
68 args->file_path = argv[++i];
69 }
70 else if (strcmp(argv[i], "-n") == 0)
71 {
72 if (i + 1 < argc)
73 args->node_id = atoi(argv[++i]);
74 }
75 else if (strcmp(argv[i], "-t") == 0)
76 {
77 if (i + 1 < argc)
78 args->n_retry = atoi(argv[++i]);
79 }
80 else if (strcmp(argv[i], "-r") == 0)
81 {
82 if (i + 1 < argc)
83 args->reset = atoi(argv[++i]);
84 }
85 else if (strcmp(argv[i], "-v") == 0)
86 {
87 if (i + 1 < argc)
88 args->verbose = atoi(argv[++i]);
89 }
90 else if (strcmp(argv[i], "-c") == 0)
91 {
92 if (i + 1 < argc)

95 else if (strcmp(argv[i], "-i") == 0)
96 {
97 if (i + 1 < argc)

100 else if (strcmp(argv[i], "-p") == 0)
101 {
102 if (i + 1 < argc)
103 args->port = atoi(argv[++i]);

105 }
106
107 /* Validate mandatory arguments */
108 if (args->file_path == NULL)
109 {
110 printf("[ ERR ] Hexfile Not Provided!\n");
111
112 return -1;
113 }
114
115 if(args->node_id == -1)
116 {
117 printf("[ ERR ] Node Number Not Provided!\n");
118
119 return -1;
120 }
121
122 return 0;
123}

137int32_t get_file_size(char* const file_name)
138{
139 int ch; /* must be int: fgetc returns EOF=-1, which is lost
140 if stored in unsigned char (default on ARM) */
141 uint32_t lines = 0;
142
143 if (file_name == NULL)
144 return 0;
145
146 /* Stat before open: lets us check file type and size before touching
147 the fd. On Raspberry Pi, fopen() on a char device (e.g. a symlink
148 to /dev/ttyAMA0) or a FIFO makes fgetc() block forever. */
149 if (stat(file_name, &hex_file_stat) < 0)
150 {
151 printf("[ ERR ] File stat info not found!\n");
152 return 0;
153 }
154
155 printf("File: %s\n", file_name);
156 printf("File Size: %lld bytes\n", (long long)hex_file_stat.st_size);
157 printf("Last modification: %s", ctime(&hex_file_stat.st_mtime));
158
159 if ((hex_file_stat.st_mode & S_IRUSR) == FLAG_RESET)
160 {
161 printf("[ ERR ] User has no file read permission!\n");
162 return 0;
163 }
164
165 if (!S_ISREG(hex_file_stat.st_mode))
166 {
167 printf("[ ERR ] Not a regular file!\n");
168 return 0;
169 }
170
171 if (hex_file_stat.st_size == 0)
172 {
173 printf("[ ERR ] File is empty (0 bytes). Nothing to flash.\n");
174 return 0;
175 }
176
177 hex_file_ptr = fopen(file_name, "r");
178 if (hex_file_ptr == NULL)
179 {
180 printf("[ ERR ] Cannot open file for reading!\n");
181 return 0;
182 }
183
184 while ((ch = fgetc(hex_file_ptr)) != EOF)
185 {
186 if (ch == '\n')
187 lines++;
188 }
189
190 fclose(hex_file_ptr);
191
192 printf("Total Lines %u\n", lines);
193 fileio_printf(&handle_log_file, "Total Lines %u\n", lines);
194 return (int32_t)lines;
195}

209int read_hex_file(char * const filename,
210 mem_pool_t * const buffer,
211 int32_t line_len,
212 int32_t * hex_base_address,
213 uint32_t * hex_end_address,
214 cmd_args_t *args)
215{
216 char line[MAX_CHAR_PER_LINE];
217 uint32_t line_count = 0;
218 uint32_t length;
219 uint32_t address, base_address = 0;
220 uint32_t type;
221 uint32_t byte_counter = 0;
222 uint8_t first_data_record_flag = 1; /* ← renamed: tracks first DATA record */
223
224 uint32_t max_records = buffer->size / sizeof(hex_record_t);
225
226 hex_record_t *record_arr = (hex_record_t *)buffer->data;
227
228 hex_file_ptr = fopen(filename, "r");
229 if (hex_file_ptr == NULL)
230 return EAGAIN;
231
232 while (fgets(line, sizeof(line), hex_file_ptr))
233 {
234 if (line[0] != ':')
235 continue;
236
237 length = hex_to_int(line + 1, 2);
238 address = hex_to_int(line + 3, 4);
239 type = hex_to_int(line + 7, 2);
240
241 record_arr[line_count].length = length;
242 record_arr[line_count].type = type;
243 record_arr[line_count].checksum = hex_to_int(line + 9 + (length * 2), 2);
244
245 /* Update the running segment base on every 0x04 record */
246 if (type == 0x04)
247 {
248 base_address = (hex_to_int(line + 9, 4) << 16);
249 }
250
251 /* Absolute address for this record */
252 record_arr[line_count].address = base_address + address;
253
254 /* Process data records */
255 if (type == 0x00)
256 {
257 uint32_t abs_addr = record_arr[line_count].address;
258
259 /* Capture hex_base_address from the FIRST data record's
260 absolute address — this is the true lowest firmware byte. */
261 if (first_data_record_flag)
262 {
263 *hex_base_address = (int32_t)abs_addr;
264 first_data_record_flag = 0;
265 }
266
267 /* Also track the minimum address in case records are not
268 in ascending order (unusual but valid in Intel HEX). */
269 if (abs_addr < (uint32_t)*hex_base_address)
270 *hex_base_address = (int32_t)abs_addr;
271
272 /* Copy data bytes */
273 for (uint32_t i = 0; i < length; i++)
274 {
275 record_arr[line_count].data[i] =
276 hex_to_int(line + 9 + (i * 2), 2);
277 byte_counter++;
278 }
279
280 /* Track the highest address seen */
281 uint32_t end_addr = abs_addr + length;
282 if (end_addr > *hex_end_address)
283 *hex_end_address = end_addr;
284 }
285
286 if (args->verbose == VERBOSE_LEVEL_3)
287 {
288 printf("**************************************\n");
289 printf("Line[%u] %s", line_count, line);
290 printf("Record[%u] %02X 0x%08X %u %02X\n",
291 line_count,
292 record_arr[line_count].type,
293 record_arr[line_count].address,
294 record_arr[line_count].length,
295 record_arr[line_count].checksum);
296 }
297
298 line_count++;
299 }
300
301 printf("Total Bytes: %u [ 0x%x -> 0x%x] \n\r",
302 byte_counter, *hex_base_address, *hex_end_address);
303
304 fclose(hex_file_ptr);
305 return 0;
306}
