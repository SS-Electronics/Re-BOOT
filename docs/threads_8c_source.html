102int thread_create(thread_t *thread, thread_func_t func, void *arg)
103{
104
105#if defined(__linux__)
106
107 if (!thread || !func)
108 {
109 return -1;
110 }
111 else
112 {
113 return pthread_create(&thread->handle, NULL, func, arg);
114 }
115
116#elif defined(_WIN32) || defined(_WIN64)
117
118 /* allocate wrapper for function + argument */
119 void **wrapper = malloc(sizeof(void*) * 2);
120
121 wrapper[0] = (void*)func;
122 wrapper[1] = arg;
123
124 if (!thread || !func || !wrapper)
125 {
126 return -1;
127 }
128 else
129 {
130 thread->handle = CreateThread(
131 NULL,
132 0,
133 win_thread_adapter,
134 wrapper,
135 0,
136 &thread->id);
137
138 return thread->handle ? 0 : -1;
139 }
140
141#endif
142}
