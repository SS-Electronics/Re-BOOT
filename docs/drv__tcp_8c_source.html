76int drv_tcp_open(drv_tcp_t *ctx,
77 const char *ip,
78 uint16_t port)
79{
80
81#if defined(_WIN32) || defined(_WIN64)
82
83 WSADATA wsa;
84 WSAStartup(MAKEWORD(2,2), &wsa);
85
86#endif
87
88 ctx->socket_fd = socket(AF_INET, SOCK_STREAM, 0);
89
90 if (ctx->socket_fd < 0)
91 return -1;
92
93 struct sockaddr_in server;
94 memset(&server, 0, sizeof(server));
95
96 server.sin_family = AF_INET;
97 server.sin_port = htons(port);
98
99#if defined(_WIN32) || defined(_WIN64)
100 server.sin_addr.s_addr = inet_addr(ip);
101#elif defined(__linux__)
102 inet_pton(AF_INET, ip, &server.sin_addr);
103#endif
104
105 if (connect(ctx->socket_fd,
106 (struct sockaddr *)&server,
107 sizeof(server)) < 0)
108 {
109 return -1;
110 }
111
112 return 0;
113}
