79int drv_udp_open(drv_udp_t *ctx,
80 const char *ip,
81 uint16_t port)
82{
83
84#if defined(_WIN32) || defined(_WIN64)
85
86 WSADATA wsa;
87 WSAStartup(MAKEWORD(2, 2), &wsa);
88
89#endif
90
91 ctx->socket_fd = socket(AF_INET, SOCK_DGRAM, 0);
92
93 if (ctx->socket_fd < 0)
94 return -1;
95
96 /* Resolve and cache the remote peer address */
97 memset(&ctx->peer, 0, sizeof(ctx->peer));
98 ctx->peer.sin_family = AF_INET;
99 ctx->peer.sin_port = htons(port);
100
101#if defined(_WIN32) || defined(_WIN64)
102 ctx->peer.sin_addr.s_addr = inet_addr(ip);
103#elif defined(__linux__)
104 if (inet_pton(AF_INET, ip, &ctx->peer.sin_addr) <= 0)
105 {
106 close(ctx->socket_fd);
107 return -1;
108 }
109#endif
110
111 /* Set receive timeout (100 ms) so drv_udp_rx() does not block
112 indefinitely — matches the behaviour of the serial driver
113 (VTIME=1 → 100 ms) and drv_tcp_rx() in non-blocking usage. */
114#if defined(_WIN32) || defined(_WIN64)
115
116 DWORD timeout_ms = 100;
117 setsockopt(ctx->socket_fd,
118 SOL_SOCKET,
119 SO_RCVTIMEO,
120 (const char *)&timeout_ms,
121 sizeof(timeout_ms));
122
123#elif defined(__linux__)
124
125 struct timeval tv;
126 tv.tv_sec = 0;
127 tv.tv_usec = 100000; /* 100 ms */
128 setsockopt(ctx->socket_fd,
129 SOL_SOCKET,
130 SO_RCVTIMEO,
131 &tv,
132 sizeof(tv));
133
134#endif
135
136 return 0;
137}

223int drv_udp_rx(drv_udp_t *ctx,
224 uint8_t *buffer,
225 uint16_t max_len)
226{
227
228#if defined(_WIN32) || defined(_WIN64)
229
230 int peer_len = sizeof(ctx->peer);
231
232#elif defined(__linux__)
233
234 socklen_t peer_len = sizeof(ctx->peer);
235
236#endif
237
238 int n = (int)recvfrom(ctx->socket_fd,
239 (char *)buffer,
240 max_len,
241 0,
242 (struct sockaddr *)&ctx->peer,
243 &peer_len);
244
245 /* Normalize timeout / EAGAIN to 0 so the transport layer's
246 "if (n <= 0) continue" loop behaves the same as for serial. */
247 if (n < 0)
248 return 0;
249
250 return n;
251}
