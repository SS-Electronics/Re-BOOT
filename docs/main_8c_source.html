102int main(int argc, char *argv[])
103{
105 printf("\n");
106 printf(" ╔════════════════════════════════════════╗\n");
107 printf(" ║ Re-BOOT Application ║\n");
108 printf(" ║ ║\n");
109 printf(" ║ Platform : ARM ║\n");
110 printf(" ║ Support : Intel HEX ║\n");
111 printf(" ║ Version : v1.0.0 ║\n");
112 printf(" ║ Developer : SS-Electronis ║\n");
113 printf(" ║ License : GPL-3.0 ║\n");
114 printf(" ║ ║\n");
115 printf(" ╚════════════════════════════════════════╝\n\n");
116
118 signal(SIGINT, handle_sigint);
119
121 status = parse_arguments(argc, argv, &cmds);
122
123 if(status == 0)
124 {
126 status = transport_init(&cmds);
127 transport_flush();
128
129 if(status != 0)
130 {
131 printf("[ ERR ] Not able to open comm port %s %s ...\n", cmds.interface, cmds.ip);
132 printf("Exiting...\n");
133
134 goto exit;
135 }
136
138 if( fileio_open(&handle_log_file, "./re-boot.log", FILEIO_WRITE) != EXIT_SUCCESS )
139 {
140 printf("[ ERR ] Log file can not be created! ...\n");
141
142 goto exit;
143 }
144
146 hex_file_lines = get_file_size(cmds.file_path);
147
148 if (hex_file_lines <= 0)
149 {
150 printf("[ ERR ] HEX file is empty or contains no data records. Exiting...\n");
151 fileio_printf(&handle_log_file, "[ ERR ] HEX file empty — abort\n");
152 goto exit;
153 }
154
156 status = create_mem_pool( &mem_pool_hex_file_head, (hex_file_lines * sizeof(hex_record_t)) );
157
158 if(status != 0 )
159 {
160 goto exit;
161 }
162
163 /* Inilialize the queues */
164 queue_init(&handle_queue_receive_packets);
165
166 if(status == 0)
167 {
168 status = read_hex_file(cmds.file_path,
169 &mem_pool_hex_file_head,
170 hex_file_lines,
171 &hex_base_address, &hex_end_address,
172 &cmds);
173 }
174
175 if( status == 0)
176 {
177 bootloader_context.records = (hex_record_t *)mem_pool_hex_file_head.data;
178 bootloader_context.record_count = (uint32_t)hex_file_lines;
179 bootloader_context.hex_base_address = (uint32_t)hex_base_address;
180 bootloader_context.hex_end_address = hex_end_address;
181 bootloader_context.node_id = cmds.node_id;
182
183 printf("Number of HEX records created : %d! ...\n", hex_file_lines);
184 fileio_printf(&handle_log_file,"Number of HEX records created : %d! ...\n", hex_file_lines);
185 }
186 else
187 {
188 printf("HEX records not created! Exiting ...\n");
189 fileio_printf(&handle_log_file,"HEX records not created! Exiting ...\n");
190
191 goto exit;
192 }
193
195 comm_thread_running = FLAG_SET;
196 thread_create(&handle_comm_rx_thread, comm_rx_thread, &comm_thread_running);
197
198 /* Init state machine */
199 fsm_init(&booloader_fsm,
200 &ST_INIT_STATE,
201 fsm_table,
202 fsm_table_size,
203 &bootloader_context);
204
205 /* Start Initialization state */
206 fsm_dispatch(&booloader_fsm, fsm_event_create(EVT_START, NULL));
207
208 while ( (booloader_fsm.fsm_running == FLAG_SET) && (app_running == FLAG_SET) )
209 {
210 act_fsm_signal_generation(&booloader_fsm);
211 fsm_run(&booloader_fsm); /* process FSM */
212 }
213 }
214 else
215 {
216 printf("Invalid Options... Exititng...\n");
217 printf("For help ebter re-boot -h\n");
218
219 }
220
221 if((app_running != FLAG_SET))
222 {
223 printf("Close Interrupt received! Exiting...\n");
224 fileio_printf(&handle_log_file,"Close Interrupt received! Exiting...\n");
225 }
226
227
229 exit:
230
231 comm_thread_running = FLAG_RESET;
232
237 transport_close(&cmds);
238
240 thread_join(&handle_comm_rx_thread);
241
242 /* Terminate the queues */
243 queue_destroy(&handle_queue_receive_packets);
244
246 free_mem_pool(&mem_pool_hex_file_head);
247
249 if(handle_log_file.handle != NULL)
250 {
251 fileio_close(&handle_log_file);
252 }
253
254 printf("Exit Done!\n");
255
256 return EXIT_SUCCESS;
257}
