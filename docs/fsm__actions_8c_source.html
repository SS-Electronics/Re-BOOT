115static void print_progress(uint32_t sectors_done,
116 uint32_t total_sectors,
117 uint32_t current_addr)
118{
119 /* Percentage based on segments for fine-grained resolution */
120 uint32_t pct = (total_segments > 0u)
121 ? (segments_sent * 100u) / total_segments
122 : 0u;
123
124 if (pct > 100u)
125 pct = 100u;
126
127 /* Build the bar string */
128 uint32_t filled = (pct * PROGRESS_BAR_WIDTH) / 100u;
129 char bar[PROGRESS_BAR_WIDTH + 1u];
130 uint32_t i;
131
132 for (i = 0u; i < PROGRESS_BAR_WIDTH; i++)
133 {
134 if (i < filled) bar[i] = '=';
135 else if (i == filled) bar[i] = '>';
136 else bar[i] = ' ';
137 }
138 bar[PROGRESS_BAR_WIDTH] = '\0';
139
140 /* Overwrite the current terminal line */
141 printf("\r Flashing [%s] %3u%% sector %u/%u 0x%08X ",
142 bar, pct, sectors_done, total_sectors, current_addr);
143
144 fflush(stdout);
145}

173void act_fsm_signal_generation(fsm_t *fsm)
174{
175 void *msg = NULL;
176
177 if (!queue_try_pop(&handle_queue_receive_packets, &msg))
178 return;
179
180 comm_packet_t *pkt = (comm_packet_t *)msg;
181 uint8_t cmd = pkt->command;
182 int node = CTX(fsm)->node_id;
183
184 /* RESP_TARGET_INFO carries the node ID in the command byte itself.
185 The expected command is a runtime value so it cannot be a case
186 label — evaluate it with an explicit comparison first. */
187 if (cmd == (uint8_t)(RESP_TARGET_INFO + node))
188 {
189 fsm_event_t *evt = fsm_event_create(EVT_TARGET_INFO, NULL);
190 evt->data = malloc(sizeof(comm_packet_t));
191 memcpy(evt->data, pkt, sizeof(comm_packet_t));
192 fsm_dispatch(fsm, evt);
193 }
194 else
195 {
196 switch (cmd)
197 {
198 case RESP_SEG_ACK:
199 fsm_dispatch(fsm, fsm_event_create(EVT_SEG_ACK, NULL));
200 break;
201
202 case RESP_CRC_ACK:
203 fsm_dispatch(fsm, fsm_event_create(EVT_CRC_OK, NULL));
204 break;
205
206 case RESP_CRC_NACK:
207 fsm_dispatch(fsm, fsm_event_create(EVT_CRC_NACK, NULL));
208 break;
209
210 case RESP_APP_JUMP_ACK:
211 fsm_dispatch(fsm, fsm_event_create(EVT_APP_ACK, NULL));
212 break;
213
214 default:
215 break;
216 }
217 }
218
219 free(pkt);
220}

301void act_target_info(fsm_event_t *e, fsm_t *fsm)
302{
303 comm_packet_t *pkt = (comm_packet_t *)e->data;
304 bootloader_ctx_t *ctx = CTX(fsm);
305
306 if (pkt == NULL)
307 {
308 printf("[ ERR ] Target info packet is NULL\n");
309 fileio_printf(&handle_log_file, "[ERR ] RESP_TARGET_INFO is NULL\n");
310 fsm->fsm_running = FLAG_RESET;
311 return;
312 }
313
314 if (pkt->length != 8)
315 {
316 printf("[ ERR ] Target info length mismatch: got %u, expected 8\n",
317 pkt->length);
318 fileio_printf(&handle_log_file,
319 "[ERR ] RESP_TARGET_INFO length=%u (expected 8)\n",
320 pkt->length);
321 free(pkt);
322 fsm->fsm_running = FLAG_RESET;
323 return;
324 }
325
326 /* Unpack target-reported flash start address */
327 uint32_t flash_addr =
328 ((uint32_t)pkt->data[0] << 24) |
329 ((uint32_t)pkt->data[1] << 16) |
330 ((uint32_t)pkt->data[2] << 8) |
331 ((uint32_t)pkt->data[3]);
332
333 uint16_t sector_size = ((uint16_t)pkt->data[4] << 8) | pkt->data[5];
334 uint16_t segment_size = ((uint16_t)pkt->data[6] << 8) | pkt->data[7];
335
336 /* ================================================================
337 ADDRESS COMPATIBILITY CHECK
338 ================================================================
339 The target reports its APP_START_ADDRESS — the lowest flash
340 address the bootloader is permitted to write.
341
342 ctx->hex_base_address comes from the HEX file parser and is the
343 lowest absolute address present in any data record.
344
345 Two rules must both pass before any data is transmitted:
346
347 Rule 1 — Non-empty image:
348 hex_end_address > hex_base_address
349 Catches an empty HEX file or a parser failure that left
350 both addresses at zero.
351
352 Rule 2 — No bootloader overlap:
353 hex_base_address >= flash_addr (target APP_START_ADDRESS)
354 If the HEX image starts below APP_START_ADDRESS it contains
355 data intended for the bootloader's own flash region.
356 Flashing it would corrupt the bootloader.
357
358 On any failure: print a clear diagnostic, halt the FSM,
359 and return — no pipeline is built, no data is sent.
360 ================================================================ */
361
362 /* Print info first so the operator always sees what was received */
363 printf("[ Re-BOOT ] Target connected (node %d)\n", ctx->node_id);
364 printf(" Flash start : 0x%08X\n", flash_addr);
365 printf(" Sector size : %u bytes\n", sector_size);
366 printf(" Segment size: %u bytes\n", segment_size);
367 printf(" HEX range : 0x%08X -> 0x%08X\n",
368 ctx->hex_base_address, ctx->hex_end_address);
369
370 fileio_printf(&handle_log_file,
371 "[TARGET] node_id=%d flash=0x%08X sector=%u B segment=%u B\n",
372 ctx->node_id, flash_addr, sector_size, segment_size);
373 fileio_printf(&handle_log_file,
374 "[TARGET] HEX range 0x%08X -> 0x%08X\n",
375 ctx->hex_base_address, ctx->hex_end_address);
376
377 /* Rule 1 — non-empty image */
378 if (ctx->hex_end_address <= ctx->hex_base_address)
379 {
380 printf("[ ERR ] HEX image is empty or has invalid address range.\n");
381 printf(" hex_base=0x%08X hex_end=0x%08X\n",
382 ctx->hex_base_address, ctx->hex_end_address);
383 printf(" Aborting.\n\n");
384
385 fileio_printf(&handle_log_file,
386 "[ERR ] Empty/invalid HEX range — abort\n");
387
388 free(pkt);
389 fsm->fsm_running = FLAG_RESET;
390 return;
391 }
392
393 /* Rule 2 — HEX image must not start below the app region */
394 if (ctx->hex_base_address < flash_addr)
395 {
396 printf("[ ERR ] HEX image address is incompatible with target.\n");
397 printf(" HEX base : 0x%08X\n", ctx->hex_base_address);
398 printf(" Target app start : 0x%08X\n", flash_addr);
399 printf(" Overlap : 0x%08X bytes into bootloader\n",
400 flash_addr - ctx->hex_base_address);
401 printf(" Aborting to protect bootloader flash.\n\n");
402
403 fileio_printf(&handle_log_file,
404 "[ERR ] Address mismatch HEX=0x%08X < target=0x%08X — abort\n",
405 ctx->hex_base_address, flash_addr);
406
407 free(pkt);
408 fsm->fsm_running = FLAG_RESET;
409 return;
410 }
411
412 /* Both checks passed */
413 printf(" Addr check : PASS "
414 "(0x%08X >= 0x%08X)\n\n",
415 ctx->hex_base_address, flash_addr);
416
417 fileio_printf(&handle_log_file,
418 "[TARGET] Addr check PASS: HEX=0x%08X >= target=0x%08X\n",
419 ctx->hex_base_address, flash_addr);
420
421 /* Store parameters and proceed */
422 ctx->sector_size = sector_size;
423 ctx->segment_size = segment_size;
424 ctx->retry_count = 0;
425 ctx->max_retries = MAX_RETRY;
426
427 free(pkt);
428
429 fsm_dispatch(fsm, fsm_event_create(EVT_START, NULL));
430}

505void act_send_window(fsm_event_t *e, fsm_t *fsm)
506{
507 bootloader_ctx_t *ctx = CTX(fsm);
508
509 uint8_t seg_data[ctx->segment_size];
510 uint32_t seg_addr = 0u;
511
512 int len = pipeline_next_segment(
513 &ctx->pipeline,
514 ctx->current_sector,
515 &ctx->offset,
516 ctx->segment_size,
517 &seg_addr,
518 seg_data
519 );
520
521 if (len < 0)
522 {
523 /* Sector exhausted — trigger CRC verify */
524 fileio_printf(&handle_log_file,
525 "[SECTOR %u] All segments sent — requesting verify\n",
526 ctx->current_sector);
527
528 fsm_dispatch(fsm, fsm_event_create(EVT_SECTOR_END, NULL));
529 return;
530 }
531
532 /* Build and send CMD_PIPELINE_DATA */
533 tx_pkt.command = CMD_PIPELINE_DATA;
534 tx_pkt.length = 4u + (uint16_t)len;
535 tx_pkt.data[0] = (seg_addr >> 24) & 0xFF;
536 tx_pkt.data[1] = (seg_addr >> 16) & 0xFF;
537 tx_pkt.data[2] = (seg_addr >> 8) & 0xFF;
538 tx_pkt.data[3] = (seg_addr >> 0) & 0xFF;
539 memcpy(&tx_pkt.data[4], seg_data, (size_t)len);
540
541 if (transport_send(&tx_pkt) > 0)
542 {
543 segments_sent++;
544
545 /* Console: update the progress bar in place */
546 uint32_t sector_base =
547 ctx->pipeline.sectors[ctx->current_sector].base_addr;
548
549 print_progress(ctx->current_sector,
550 ctx->pipeline.sector_count,
551 sector_base);
552
553 /* Log: full segment record */
554 fileio_printf(&handle_log_file,
555 "[SEG TX ] sector=%u addr=0x%08X len=%3d B data=[%02X %02X %02X %02X ...] total_sent=%u\n",
556 ctx->current_sector, seg_addr, len,
557 (len > 0) ? seg_data[0] : 0u,
558 (len > 1) ? seg_data[1] : 0u,
559 (len > 2) ? seg_data[2] : 0u,
560 (len > 3) ? seg_data[3] : 0u,
561 segments_sent);
562 }
563 else
564 {
565 /* Print on a new line so the error does not clobber the bar */
566 printf("\n[ ERR ] Segment send failed at 0x%08X\n", seg_addr);
567 fileio_printf(&handle_log_file,
568 "[ERR ] SEG TX failed sector=%u addr=0x%08X\n",
569 ctx->current_sector, seg_addr);
570 }
571}

679void act_crc_nack(fsm_event_t *e, fsm_t *fsm)
680{
681 bootloader_ctx_t *ctx = CTX(fsm);
682
683 ctx->retry_count++;
684
685 printf(" [ WARN ] Sector %u NACK — retry %u / %u\n",
686 ctx->current_sector, ctx->retry_count, ctx->max_retries);
687
688 fileio_printf(&handle_log_file,
689 "[NACK ] sector=%u retry=%u/%u\n",
690 ctx->current_sector, ctx->retry_count, ctx->max_retries);
691
692 if (ctx->retry_count >= ctx->max_retries)
693 {
694 printf("[ ERR ] Max retries reached on sector %u — aborting\n",
695 ctx->current_sector);
696 fileio_printf(&handle_log_file,
697 "[ERR ] Sector %u: max retries (%u) exceeded — abort\n",
698 ctx->current_sector, ctx->max_retries);
699
700 fsm->fsm_running = FLAG_RESET;
701 return;
702 }
703
704 /* Rewind sector offset so retransmission starts from byte 0 */
705 ctx->offset = 0u;
706
707 /* Rewind the segment counter so the progress bar stays accurate */
708 uint32_t segs_per_sector = (ctx->sector_size + ctx->segment_size - 1u)
709 / ctx->segment_size;
710
711 if (segments_sent >= segs_per_sector)
712 segments_sent -= segs_per_sector;
713 else
714 segments_sent = 0u;
715
716 fileio_printf(&handle_log_file,
717 "[NACK ] Retransmitting sector %u from byte 0\n",
718 ctx->current_sector);
719
720 fsm_dispatch(fsm, fsm_event_create(EVT_START, NULL));
721}
