223static uint16_t crc16_ccitt(const uint8_t *data, uint16_t len)
224{
225 uint16_t crc = 0xFFFF;
226
227 for (uint16_t i = 0; i < len; i++)
228 {
229 crc ^= (uint16_t)data[i] << 8;
230
231 for (uint8_t j = 0; j < 8u; j++)
232 {
233 if (crc & 0x8000u)
234 crc = (crc << 1) ^ 0x1021u;
235 else
236 crc <<= 1;
237 }
238 }
239
240 return crc;
241}

265int transport_init(cmd_args_t *cmds)
266{
267 if (cmds->interface == NULL)
268 {
269 printf("[ ERR ] Communication interface not provided!\n");
270 printf("Use: '-c serial', '-c tcp', '-c udp', or '-c can'\n");
271 return -1;
272 }
273
274 if (strcmp(cmds->interface, "serial") == 0)
275 {
276 if (cmds->ip == NULL)
277 {
278 printf("[ ERR ] Serial port not specified.\n");
279 printf("Use: '-i ttyACM0' or '-i COM15'\n");
280 return -1;
281 }
282
283 driver_type = SERIAL;
284 return drv_serial_open(&handle_serial_driver, cmds->ip, 115200);
285 }
286 else if (strcmp(cmds->interface, "tcp") == 0)
287 {
288 if (cmds->ip == NULL)
289 {
290 printf("[ ERR ] TCP IP/port not specified.\n");
291 printf("Use: '-i 192.168.0.1 -p 5000'\n");
292 return -1;
293 }
294
295 driver_type = TCP;
296 return drv_tcp_open(&handle_tcp_driver, cmds->ip, cmds->port);
297 }
298 else if (strcmp(cmds->interface, "udp") == 0)
299 {
300 if (cmds->ip == NULL)
301 {
302 printf("[ ERR ] UDP IP/port not specified.\n");
303 printf("Use: '-i 192.168.0.1 -p 5000'\n");
304 return -1;
305 }
306
307 driver_type = UDP;
308 return drv_udp_open(&handle_udp_driver, cmds->ip, cmds->port);
309 }
310 else if (strcmp(cmds->interface, "can") == 0)
311 {
312 if (cmds->ip == NULL)
313 {
314 printf("[ ERR ] CAN adapter port not specified.\n");
315 printf("Use: '-i ttyUSB0' (the USB-CAN adapter virtual port)\n");
316 return -1;
317 }
318
319 /* CAN reuses the serial driver — adapter appears as a virtual COM port.
320 Frame packing/parsing is different from the serial path. */
321 driver_type = CAN;
322 return drv_serial_open(&handle_serial_driver, cmds->ip, 115200);
323 }
324 else
325 {
326 printf("[ ERR ] Unknown interface '%s'.\n", cmds->interface);
327 printf("Use: '-c serial', '-c tcp', '-c udp', or '-c can'\n");
328 return -1;
329 }
330}

385int transport_send(comm_packet_t *pkt)
386{
387 if (!pkt)
388 return -1;
389
390 /* ----------------------------------------------------------------
391 CAN-over-Serial framing
392 ':' | CMD(1B) | LEN(1B) | DATA[0..LEN-1]
393 Total: 3 + LEN bytes (max 11 bytes for LEN=8)
394 ---------------------------------------------------------------- */
395 if (driver_type == CAN)
396 {
397 if (pkt->length > CAN_MAX_PAYLOAD)
398 {
399 printf("[ ERR ] CAN payload %u > CAN_MAX_PAYLOAD (%u)"
400 " — truncating\n", pkt->length, CAN_MAX_PAYLOAD);
401 }
402
403 uint8_t len = (pkt->length <= CAN_MAX_PAYLOAD)
404 ? (uint8_t)pkt->length
405 : (uint8_t)CAN_MAX_PAYLOAD;
406
407 /* Build the frame in a local buffer */
408 uint8_t frame[3u + CAN_MAX_PAYLOAD];
409 uint8_t pos = 0u;
410
411 frame[pos++] = ':'; /* start delimiter — same as Serial */
412 frame[pos++] = pkt->command; /* CMD = CAN ID (low 8 bits) */
413 frame[pos++] = len; /* LEN = single byte (0–8) */
414
415 if (len > 0u)
416 {
417 memcpy(&frame[pos], pkt->data, len);
418 pos += len;
419 }
420
421 /* No CRC — CAN hardware handles frame integrity */
422 return drv_serial_tx(&handle_serial_driver, frame, pos);
423 }
424
425 /* ----------------------------------------------------------------
426 Serial / TCP framing (byte-stream with CRC16)
427 ':' | CMD | LEN_H | LEN_L | DATA | CRC_H | CRC_L
428 ---------------------------------------------------------------- */
429 if (pkt->length > 2048u - 6u)
430 return -1;
431
432 uint8_t frame[2048];
433 uint16_t pos = 0;
434
435 frame[pos++] = ':';
436 frame[pos++] = pkt->command;
437 frame[pos++] = (pkt->length >> 8) & 0xFF; /* LEN_H */
438 frame[pos++] = pkt->length & 0xFF; /* LEN_L */
439
440 if (pkt->length > 0u)
441 {
442 memcpy(&frame[pos], pkt->data, pkt->length);
443 pos += pkt->length;
444 }
445
446 /* CRC16-CCITT over CMD + LEN_H + LEN_L + DATA */
447 uint16_t crc = crc16_ccitt(&frame[1], (uint16_t)(pkt->length + 3u));
448
449 frame[pos++] = (crc >> 8) & 0xFF;
450 frame[pos++] = crc & 0xFF;
451
452 if (driver_type == SERIAL)
453 return drv_serial_tx(&handle_serial_driver, frame, pos);
454
455 if (driver_type == TCP)
456 return drv_tcp_tx(&handle_tcp_driver, frame, pos);
457
458 if (driver_type == UDP)
459 return drv_udp_tx(&handle_udp_driver, frame, pos);
460
461 return -1;
462}

501int transport_receive(comm_packet_t *pkt, int32_t *thread_running_flag)
502{
503 if (!pkt || !thread_running_flag)
504 return -1;
505
506 uint8_t byte;
507
508 while (*thread_running_flag)
509 {
510 /* Read one byte from the active driver */
511 int n = -1;
512
513 if (driver_type == SERIAL || driver_type == CAN)
514 n = drv_serial_rx(&handle_serial_driver, &byte, 1);
515 else if (driver_type == TCP)
516 n = drv_tcp_rx(&handle_tcp_driver, &byte, 1);
517 else if (driver_type == UDP)
518 n = drv_udp_rx(&handle_udp_driver, &byte, 1);
519
520 if (n <= 0)
521 continue;
522
523 /* ============================================================
524 CAN-over-Serial parser
525 Frame: ':' | CMD(1B) | LEN(1B) | DATA[0..LEN-1]
526 ============================================================ */
527 if (driver_type == CAN)
528 {
529 switch (can_rx_state)
530 {
531 /* --------------------------------------------------------
532 State 0: scan for ':' start-of-frame delimiter.
533 Discarding non-':' bytes provides the same self-healing
534 re-sync as the Serial/TCP parser.
535 -------------------------------------------------------- */
536 case 0:
537 if (byte == (uint8_t)':')
538 can_rx_state = 1;
539 break;
540
541 /* --------------------------------------------------------
542 State 1: CMD byte.
543 Carries the CAN command code (= CAN ID low 8 bits).
544 -------------------------------------------------------- */
545 case 1:
546 pkt->command = byte;
547 can_rx_state = 2;
548 break;
549
550 /* --------------------------------------------------------
551 State 2: LEN byte — single byte replaces LEN_H:LEN_L.
552 Validate against CAN_MAX_PAYLOAD (8).
553 -------------------------------------------------------- */
554 case 2:
555 if (byte > CAN_MAX_PAYLOAD)
556 {
557 printf("[ RX CAN ] LEN %u > CAN_MAX_PAYLOAD (%u)"
558 " — discarding frame\n",
559 byte, CAN_MAX_PAYLOAD);
560 can_parser_reset();
561 return -3;
562 }
563
564 can_rx_len = byte;
565 pkt->length = byte;
566 can_rx_index = 0u;
567
568 /* Zero-length CAN frame — valid, no data phase needed */
569 if (can_rx_len == 0u)
570 {
571 can_parser_reset();
572 return 0;
573 }
574
575 can_rx_state = 3;
576 break;
577
578 /* --------------------------------------------------------
579 State 3: read exactly LEN data bytes.
580 Return as soon as the last byte is stored.
581 -------------------------------------------------------- */
582 case 3:
583 pkt->data[can_rx_index++] = byte;
584
585 if (can_rx_index == can_rx_len)
586 {
587 uint8_t pkt_len = can_rx_len;
588 can_parser_reset();
589 return (int)pkt_len;
590 }
591 break;
592
593 default:
594 can_parser_reset();
595 break;
596 }
597
598 continue; /* back to top of while loop */
599 }
600
601 /* ============================================================
602 Serial / TCP parser
603 Frame: ':' | CMD | LEN_H | LEN_L | DATA | CRC_H | CRC_L
604 ============================================================ */
605 switch (rx_state)
606 {
607 /* ----------------------------------------------------------
608 State 0: scan for ':' start-of-frame delimiter.
609 ---------------------------------------------------------- */
610 case 0:
611 if (byte == (uint8_t)':')
612 rx_state = 1;
613 break;
614
615 /* ----------------------------------------------------------
616 State 1: CMD byte.
617 ---------------------------------------------------------- */
618 case 1:
619 pkt->command = byte;
620 rx_state = 2;
621 break;
622
623 /* ----------------------------------------------------------
624 State 2: LEN_H — high byte of payload length.
625 ---------------------------------------------------------- */
626 case 2:
627 rx_length = (uint16_t)((uint16_t)byte << 8);
628 rx_state = 3;
629 break;
630
631 /* ----------------------------------------------------------
632 State 3: LEN_L — low byte of payload length.
633 Skip data phase for zero-length packets.
634 ---------------------------------------------------------- */
635 case 3:
636 rx_length |= (uint16_t)byte;
637
638 if (rx_length > (uint16_t)sizeof(pkt->data))
639 {
640 printf("[ RX ] Frame error: length %u exceeds buffer"
641 " %u — discarding\n",
642 rx_length, (uint16_t)sizeof(pkt->data));
643 parser_reset();
644 return -3;
645 }
646
647 pkt->length = rx_length;
648 rx_index = 0;
649 rx_state = (rx_length == 0u) ? 5u : 4u;
650 break;
651
652 /* ----------------------------------------------------------
653 State 4: accumulate data bytes.
654 ---------------------------------------------------------- */
655 case 4:
656 pkt->data[rx_index++] = byte;
657
658 if (rx_index == rx_length)
659 rx_state = 5;
660 break;
661
662 /* ----------------------------------------------------------
663 State 5: CRC_H byte.
664 ---------------------------------------------------------- */
665 case 5:
666 rx_crc = (uint16_t)((uint16_t)byte << 8);
667 rx_state = 6;
668 break;
669
670 /* ----------------------------------------------------------
671 State 6: CRC_L byte — validate and return.
672 Save rx_crc to a local BEFORE parser_reset() zeroes it.
673 ---------------------------------------------------------- */
674 case 6:
675 {
676 rx_crc |= (uint16_t)byte;
677
678 uint16_t received_crc = rx_crc; /* save before reset */
679
680 uint8_t crc_buf[COMM_MAX_DATA + 3u];
681 uint16_t crc_pos = 0u;
682
683 crc_buf[crc_pos++] = pkt->command;
684 crc_buf[crc_pos++] = (pkt->length >> 8) & 0xFF;
685 crc_buf[crc_pos++] = pkt->length & 0xFF;
686
687 if (pkt->length > 0u)
688 {
689 memcpy(&crc_buf[crc_pos], pkt->data, pkt->length);
690 crc_pos += pkt->length;
691 }
692
693 uint16_t crc_calc = crc16_ccitt(crc_buf, crc_pos);
694 uint16_t pkt_len = pkt->length;
695
696 parser_reset(); /* all needed values are in locals now */
697
698 if (crc_calc != received_crc)
699 {
700 printf("[ RX ] CRC mismatch: calc=0x%04X rx=0x%04X"
701 " cmd=0x%02X — discarding\n",
702 crc_calc, received_crc, pkt->command);
703 return -5;
704 }
705
706 return (int)pkt_len;
707 }
708
709 default:
710 parser_reset();
711 break;
712 }
713 }
714
715 return -1;
716}

733int transport_flush(void)
734{
735 uint8_t tmp;
736
737 if (driver_type == SERIAL || driver_type == CAN)
738 {
739 while (drv_serial_rx(&handle_serial_driver, &tmp, 1) > 0)
740 {
741 /* discard stale bytes */
742 }
743 }
744 else if (driver_type == TCP)
745 {
746 while (drv_tcp_rx(&handle_tcp_driver, &tmp, 1) > 0)
747 {
748 /* discard stale bytes */
749 }
750 }
751 else if (driver_type == UDP)
752 {
753 while (drv_udp_rx(&handle_udp_driver, &tmp, 1) > 0)
754 {
755 /* discard stale datagrams */
756 }
757 }
758
759 parser_reset();
760 can_parser_reset();
761
762 return 0;
763}
